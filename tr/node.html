Clean API

A preview of the simple, intuitive API

Terminal 
# Install from npm
npm install flyto2

# Or with yarn
yarn add flyto2

# Or with pnpm
pnpm add flyto2
index.ts 
import { flyto } from 'flyto2';

// Execute a module
const result = await flyto.execute(
  'string.uppercase',
  { text: 'hello world' }
);

console.log(result.data);
// Output: "HELLO WORLD"
